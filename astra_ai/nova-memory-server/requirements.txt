from fastapi import FastAPI, HTTPException
from nova_memory_interface import NovaMemoryInterface
from enhanced_nova_memory_interface import EnhancedNovaMemoryInterface
from memory_cleanup_system import MemoryCleanupSystem
import logging

app = FastAPI()

# Initialize memory interfaces
memory_interface = EnhancedNovaMemoryInterface()
cleanup_system = MemoryCleanupSystem(memory_file="nova_ai_memory.json")

@app.on_event("startup")
async def startup_event():
    # Initialize memory systems
    memory_interface.initialize()
    cleanup_system.start_automatic_cleanup()

@app.on_event("shutdown")
async def shutdown_event():
    cleanup_system.stop_automatic_cleanup()

@app.post("/process_conversation/")
async def process_conversation(user_message: str, ai_response: str):
    result = memory_interface.process_conversation(user_message, ai_response)
    if not result.get("success"):
        raise HTTPException(status_code=500, detail=result.get("error"))
    return result

@app.get("/get_context/")
async def get_context(context_type: str = "comprehensive"):
    context = memory_interface.get_context_for_ai_response(context_type)
    return context

@app.get("/memory_statistics/")
async def memory_statistics():
    stats = memory_interface.get_memory_statistics()
    return stats

@app.post("/cleanup/")
async def run_cleanup(policy: str = "balanced"):
    cleanup_stats = cleanup_system.cleanup_memories(policy)
    return cleanup_stats
